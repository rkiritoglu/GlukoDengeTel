    async function loadChartData() {

      const querySnapshot = await getDocs(q);
      
      let allDocs = [];
      querySnapshot.forEach((doc) => {
        allDocs.push({ id: doc.id, data: doc.data() });
      });

      // Tarihe göre (ID) azalan sırada (en yeni en üstte) sırala
      allDocs.sort((a, b) => b.id.localeCompare(a.id));

      // Sınırlandırma (Limit) uygula
      if (currentChartFilter !== 'all') {
        const limitVal = parseInt(currentChartFilter);
        allDocs = allDocs.slice(0, limitVal);
      }
      
      let labels = [];
      let sugarData = [];
      let compData = [];

      allDocs.forEach((docObj) => {
        const data = docObj.data;
        const dateId = docObj.id; // YYYY-MM-DD
        
        // Yeni dinamik egzersiz sayısını al
        const exercisesArr = data.exercises || [];
        const totalTasks = 1 + exercisesArr.length;
        
        const tasksObj = data.tasks || {};
        const completed = Object.values(tasksObj).filter(Boolean).length;
        const compPercent = totalTasks > 0 ? Math.round((completed / totalTasks) * 100) : 0;

        // O günkü en yüksek şeker değerini (veya son değeri) al
        let maxSugar = null;
        if(data.sugars && data.sugars.length > 0) {
          maxSugar = data.sugars[data.sugars.length - 1].val; // Son ölçümü grafiğe yansıt
        }

        labels.push(dateId.split('-').slice(1).join('/')); // MM/DD formatı
        sugarData.push(maxSugar);
        compData.push(compPercent);
      });

      // Tarihleri eski->yeni sırasına çevir
      labels.reverse();
      sugarData.reverse();
      compData.reverse();

      renderChart(labels, sugarData, compData);
    }

    function renderChart(labels, sugarData, compData) {
      if (chartInstance) chartInstance.destroy();
      const ctx = document.getElementById('healthChart').getContext('2d');
      
      chartInstance = new Chart(ctx, {
        type: 'line',
        data: {
          labels: labels,
          datasets: [
            {
              label: 'Şeker (mg/dL)',
              data: sugarData,
              borderColor: '#ef4444',
              backgroundColor: 'rgba(239, 68, 68, 0.2)',
              yAxisID: 'ySugar',
              tension: 0.3,
              spanGaps: true
            },
            {
              label: 'Görev (%)',
              data: compData,
              borderColor: '#10b981',
              backgroundColor: 'rgba(16, 185, 129, 0.15)',
              yAxisID: 'yComp',
              type: 'bar',
              borderRadius: 4
            }
          ]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          scales: {
            x: { grid: { display: false }, ticks: { color: '#64748b' } },
            ySugar: { position: 'left', ticks: { color: '#ef4444' }, grid: { color: '#334155' } },
            yComp: { position: 'right', min: 0, max: 100, ticks: { color: '#10b981', callback: v => '%' + v }, grid: { display: false } }
          },
          plugins: { legend: { labels: { color: '#94a3b8' } } }
        }
      });
    }
